run this commands on u-boot 




# 1. Inspect files on the virtual block drive
fatls virtio 0

# 2. Load the Linux kernel Image into RAM at address 0x40400000
fatload virtio 0 0x40400000 Image

# 3. Load your initramfs.cpio.gz into RAM at address 0x44000000
fatload virtio 0 0x44000000 initramfs.cpio.gz

# 4. Set the kernel serial console arguments
setenv bootargs "console=ttyAMA0 earlycon rdinit=/init"

# 5. Boot the ARM64 kernel (Format: booti <kernel_addr> <initrd_addr>:<size> <dtb_addr>)
booti 0x40400000 0x44000000:$filesize $fdtcontroladdr
